\begin{enumerate}[label=\thesubsection.\arabic*, ref=\thesubsection.\theenumi]
%---------------- Functions and matrices
\item Domain of inverse functions. $\cos^{-1}x$ and $\sin^{-1}x$ are defined only for $-1 \le x \le 1$, while $\sin x$ and $\cos x$ are defined for all real $x$. The domain of a sum is the intersection of the domains. 
\item Counting. If each of $m$ independent positions can take $c$ values, the total number of arrangements is
\begin{align}
	N = c^m \label{eq:CBSE12-count}
\end{align}
\item Symmetric matrix. A square matrix is symmetric if
\begin{align}
	\vec{A}^{\top} = \vec{A} \label{eq:CBSE12-symmetric}
\end{align}
Every diagonal matrix is symmetric.
\item Determinant properties. For $n\times n$ matrices $\vec{A}$, $\vec{B}$ and a scalar $k$,
\begin{align}
	\abs{k\vec{A}} = k^n\abs{\vec{A}}, \qquad \abs{\vec{A}\vec{B}} = \abs{\vec{A}}\abs{\vec{B}} \label{eq:CBSE12-det}
\end{align}
\item Adjugate. With $\vec{A}\,\text{adj}\,\vec{A} = \abs{\vec{A}}\vec{I}$,
\begin{align}
	\abs{\text{adj}\,\vec{A}} = \abs{\vec{A}}^{n-1} \label{eq:CBSE12-adj}
\end{align}
\solution
Taking determinants of $\vec{A}\,\text{adj}\,\vec{A} = \abs{\vec{A}}\vec{I}$ gives $\abs{\vec{A}}\abs{\text{adj}\,\vec{A}} = \abs{\vec{A}}^n$. 
\item Characteristic polynomial and Cayley--Hamilton theorem. The characteristic polynomial of a square matrix is $f\brak{\lambda} = \abs{\vec{A} - \lambda\vec{I}}$, and
\begin{align}
	\abs{\vec{A}} = f\brak{0} = \lambda_1\cdots\lambda_n, \qquad f\brak{\vec{A}} = \vec{0} \label{eq:CBSE12-ch}
\end{align}
For a $2\times2$ matrix, $\vec{A}^2 - \text{tr}\brak{\vec{A}}\vec{A} + \abs{\vec{A}}\vec{I} = \vec{0}$; the eigenvalues sum to the trace and multiply to the determinant. For a $3\times3$ matrix with $f\brak{\lambda} = -\lambda^3 + s_1\lambda^2 - s_2\lambda + s_3$, Cayley--Hamilton gives
\begin{align}
	\text{adj}\,\vec{A} = \vec{A}^2 - s_1\vec{A} + s_2\vec{I} \label{eq:CBSE12-adjch}
\end{align}
\item Gaussian elimination and rank. For $\vec{A}\vec{x} = \vec{b}$ with $n$ unknowns, row-reduce the augmented matrix $\myvec{\vec{A} & \vec{b}}$. The system is consistent if and only if
\begin{align}
	\rank\brak{\vec{A}} = \rank\myvec{\vec{A} & \vec{b}} \label{eq:CBSE12-rank}
\end{align}
and the solution is unique when this common rank equals $n$.
%---------------- Calculus
\item Monotonicity. If $f^{\prime}\brak{x} > 0$ for all $x$ in an interval, $f$ is strictly increasing there; if $f^{\prime}\brak{x} < 0$, it is strictly decreasing. 
\item Trapezoidal rule. With $n$ subintervals of width $h = \frac{b-a}{n}$ and nodes $x_i = a + ih$,
\begin{align}
	\int_a^b f\brak{x}\,dx \approx T_n = \frac{h}{2}\brak{f\brak{x_0} + 2\sum_{i=1}^{n-1}f\brak{x_i} + f\brak{x_n}} \label{eq:CBSE12-trap}
\end{align}
The error is $-\frac{\brak{b-a}h^2}{12}f^{\prime\prime}\brak{\xi}$ for some $\xi \in \brak{a,b}$, so $T_n$ converges to the integral as $n \to \infty$.
\item Trapezoidal rule as a difference equation. With $I\sbrak{n} = \int_a^{x_n} f\brak{x}\,dx$ and $I\sbrak{0} = 0$,
\begin{align}
	I\sbrak{n+1} = I\sbrak{n} + \frac{h}{2}\brak{f\brak{x_n} + f\brak{x_{n+1}}} \label{eq:CBSE12-trapdiff}
\end{align}
and the integral is $I\sbrak{N}$.
\item Derivatives. Product rule, chain rule and the exponential rule,
\begin{align}
	\frac{d}{dx}\brak{uv} = u^{\prime}v + uv^{\prime}, \quad
	\frac{d}{dx}a^x = a^x\ln a, \quad
	\frac{d}{dx}\log\brak{\sin u} = \cot u\,\frac{du}{dx} \label{eq:CBSE12-deriv}
\end{align}

\item Symmetric integral. If $g\brak{\pi - x} = g\brak{x}$,
\begin{align}
	\int_0^\pi x\,g\brak{x}\,dx = \frac{\pi}{2}\int_0^\pi g\brak{x}\,dx \label{eq:CBSE12-king}
\end{align}
\solution
Substituting $x \to \pi - x$ gives $I = \int_0^\pi\brak{\pi - x}g\brak{x}\,dx$. Adding this to the original, $2I = \pi\int_0^\pi g\brak{x}\,dx$.
\item Laplace transform. For a first-order equation $y^{\prime} + ay = f\brak{x}$,
\begin{align}
	\mathcal{L}\cbrak{y^{\prime}} = sY\brak{s} - y\brak{0}, \quad \mathcal{L}\cbrak{\cos x} = \frac{s}{s^2+1}, \quad \mathcal{L}\cbrak{e^{-ax}} = \frac{1}{s+a} \label{eq:CBSE12-laplace}
\end{align}
\begin{align}
	\mathcal{L}\cbrak{1} = \frac{1}{s}, \quad \mathcal{L}\cbrak{t} = \frac{1}{s^2} \label{eq:CBSE12-laplace2}
\end{align}
The order of a differential equation is the highest derivative present; the degree is its power. 
%---------------- Vectors and 3D geometry
\item Orthogonality and scalar product with a unit vector,
\begin{align}
	\abs{\vec{a}+\vec{b}} = \abs{\vec{a}-\vec{b}} \iff \vec{a}^{\top}\vec{b} = 0, \qquad
	\text{scalar product of } \vec{a} \text{ with unit vector along } \vec{v} = \frac{\vec{a}^{\top}\vec{v}}{\norm{\vec{v}}} \label{eq:CBSE12-vec}
\end{align}
\item Shortest distance between lines $\vec{x} = \vec{A} + \kappa_1\vec{m}_1$ and $\vec{x} = \vec{B} + \kappa_2\vec{m}_2$. If $\text{rank}\myvec{\vec{m}_1 & \vec{m}_2} = 1$ the lines are parallel and the distance is the least value of $\norm{\vec{A} + \kappa\vec{m}_1 - \vec{B}}$, obtained from the normal equation
\begin{align}
	\vec{m}_1^{\top}\vec{m}_1\,\kappa = \vec{m}_1^{\top}\brak{\vec{B}-\vec{A}} \label{eq:CBSE12-pardist}
\end{align}
For skew lines, $\vec{M} = \myvec{\vec{m}_1 & \vec{m}_2}$ and the least squares solution of $\vec{M}\vec{\kappa} = \vec{B} - \vec{A}$ is used.
\item Points on a line in space. A line $\vec{x} = \vec{A} + \kappa\vec{m}$ is written with one parameter $\kappa$, and a point of it is at distance $r$ from $\vec{P}$ when $\norm{\vec{A} + \kappa\vec{m} - \vec{P}}^2 = r^2$, a quadratic in $\kappa$.
\item Foot of the perpendicular and image. For a point $\vec{P}$ and the line $\vec{A} + \kappa\vec{m}$,
\begin{align}
	\kappa_0 = \frac{\vec{m}^{\top}\brak{\vec{P}-\vec{A}}}{\vec{m}^{\top}\vec{m}}, \quad \vec{H} = \vec{A} + \kappa_0\vec{m}, \quad \vec{P}^{\prime} = 2\vec{H} - \vec{P} \label{eq:CBSE12-image}
\end{align}
%---------------- Linear programming
\item Corner point theorem. The maximum or minimum of a linear objective over a feasible region occurs at a corner point. For an unbounded region, a minimum exists only if the objective is bounded below there. 
\item Linear programming in standard form. ``Maximise $\vec{c}^{\top}\vec{x}$ subject to $\vec{A}\vec{x} \preceq \vec{b}$, $\vec{x} \succeq \vec{0}$'' (or minimise with $\succeq$) is solved numerically with \texttt{cvxpy}.
%---------------- Probability
\item Conditional probability,
\begin{align}
	\pr{A\mid B} = \frac{\pr{A\cap B}}{\pr{B}}, \qquad \pr{A^{\prime}\cap B} = \pr{B} - \pr{A\cap B} \label{eq:CBSE12-cond}
\end{align}
and $\pr{A\cap B} \le \min\cbrak{\pr{A},\pr{B}}$. 
\item Probability from a pmf. For a random variable $X$ with pmf $p_X\brak{x}$ and an event $E$,
\begin{align}
	\pr{E} = \sum_{x \in E} p_X\brak{x}, \quad \sum_x p_X\brak{x} = 1, \quad E\sbrak{X} = \sum_x x\,p_X\brak{x} \label{eq:CBSE12-pmf}
\end{align}
For independent $X$ and $Y$ the joint pmf is $p_{XY}\brak{x,y} = p_X\brak{x}p_Y\brak{y}$.
\item Total probability and Bayes' theorem. For mutually exclusive, exhaustive events $A_1, \dots, A_n$,
\begin{align}
	\pr{D} = \sum_i \pr{D\mid A_i}\pr{A_i}, \qquad
	\pr{A_1\mid D} = \frac{\pr{D\mid A_1}\pr{A_1}}{\pr{D}} \label{eq:CBSE12-bayes}
\end{align}
\end{enumerate}
